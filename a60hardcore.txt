local function playRandomBreakSound(position, volume)
    local soundPart = Instance.new("Part")
    soundPart.Size = Vector3.new(1, 1, 1)
    soundPart.Position = position
    soundPart.Anchored = true
    soundPart.CanCollide = false
    soundPart.Transparency = 1
    soundPart.Parent = workspace

    local randomSoundId = breakSounds[math.random(1, #breakSounds)]

    local sound = Instance.new("Sound")
    sound.SoundId = randomSoundId
    sound.Volume = volume or 5.0
    sound.RollOffMaxDistance = 250
    sound.RollOffMinDistance = 20
    sound.Parent = soundPart

    sound:Play()
    Debris:AddItem(soundPart, 3)
end

---====== Check Doors Hiding System ======---

local function isPlayerHiding()
    local player = Players.LocalPlayer
    if not player or not player.Character then return false end

    local char = player.Character
    if char:GetAttribute("Hiding") or char:GetAttribute("Hidden") then
        return true
    end

    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if humanoid and humanoid.SeatPart then
        local seatName = humanoid.SeatPart.Parent.Name:lower()
        if string.find(seatName, "wardrobe") or string.find(seatName, "closet") or string.find(seatName, "bed") or string.find(seatName, "locker") then
            return true
        end
    end

    return false
end

local BLACKLIST = {
    "Wardrobe",
    "Closet",
    "Locker",
    "Cabinet",
    "Bed",
    "Carpet",
    "Rug",
    "Figure",
    "Baseplate",

local PLANT_KEYWORDS = {
    "plant",
    "pot",
    "potted",
    "vase",
    "flower",
    "leaf",
    "leaves",
    "bush"
}

local function isPlant(instance)
    if not instance then return false end
    local nameLower = instance.Name:lower()
    for _, kw in ipairs(PLANT_KEYWORDS) do
        if string.find(nameLower, kw) then
            return true
        end
    end
    if instance.Parent and instance.Parent ~= workspace then
        local parentLower = instance.Parent.Name:lower()
        for _, kw in ipairs(PLANT_KEYWORDS) do
            if string.find(parentLower, kw) then
                return true
            end
        end
    end
    return false
end

local function isHidingSpotStructure(instance)
    if not instance then return false end
    local current = instance
    while current and current ~= workspace do
        local nameLower = current.Name:lower()
        if string.find(nameLower, "wardrobe") or 
           string.find(nameLower, "closet") or 
           string.find(nameLower, "locker") or 
           string.find(nameLower, "cabinet") or 
           string.find(nameLower, "bed") then
            return true
        end
        current = current.Parent
    end
    return false
end

local function isBlacklisted(instance)
    if not instance then return true end

    if isHidingSpotStructure(instance) then
        return true
    end

    if isPlant(instance) then
        return false
    end

    for _, name in ipairs(BLACKLIST) do
        if instance.Name == name or string.find(instance.Name:lower(), name:lower()) then
            return true
        end
    end

    if instance.Parent and instance.Parent ~= workspace then
        for _, name in ipairs(BLACKLIST) do
            if instance.Parent.Name == name or string.find(instance.Parent.Name:lower(), name:lower()) then
                return true
            end
        end
    end

    return false
end

local function isGuidingLightBook(instance)
    if not instance then return false end
    local nameLower = instance.Name:lower()
    if nameLower == "livehintbook" or string.find(nameLower, "guiding") or instance:GetAttribute("IsGuidingLight") then
        return true
    end
    return false
end

local function processBookshelf(model, moveDirection)
    local primaryPart = model.PrimaryPart or model:FindFirstChildWhichIsA("BasePart")
    if not primaryPart then return end

    if not model:GetAttribute("A60Tipped") then
        model:SetAttribute("A60Tipped", true)
        playRandomBreakSound(primaryPart.Position, 7.0)
    end

    for _, desc in ipairs(model:GetDescendants()) do
        if desc:IsA("BasePart") then
            desc.Anchored = false
        end
    end

    local throwVelocity = (moveDirection + Vector3.new(0, 0.25, 0)).Unit * math.random(120, 180)
    primaryPart.AssemblyLinearVelocity = throwVelocity
end

local warningGuiActive = false

local function showEntityWarning()
    local player = Players.LocalPlayer
    if not player then return end
    local playerGui = player:FindFirstChild("PlayerGui")
    if not playerGui or playerGui:FindFirstChild("A60WarningGui") then return end

    local screenGui = Instance.new("ScreenGui")
    screenGui.Name = "A60WarningGui"
    screenGui.IgnoreGuiInset = true
    screenGui.DisplayOrder = 9998
    screenGui.Parent = playerGui

    local textLabel = Instance.new("TextLabel")
    textLabel.Size = UDim2.new(1, 0, 0, 50)
    textLabel.Position = UDim2.new(0, 0, 0.15, 0)
    textLabel.BackgroundTransparency = 1
    textLabel.TextColor3 = Color3.fromRGB(255, 50, 50)
    textLabel.TextScaled = true
    textLabel.Font = Enum.Font.SourceSansBold
    textLabel.Text = " "
    textLabel.Parent = screenGui

    task.delay(4, function()
        if screenGui and screenGui.Parent then screenGui:Destroy() end
    end)
end

RunService.Stepped:Connect(function()
    if not entity.Model then 
        warningGuiActive = false
        return 
    end

    local model = entity.Model
    local root = model.PrimaryPart or model:FindFirstChildWhichIsA("BasePart")
    local player = Players.LocalPlayer
    if not root or not player or not player.Character then return end

    local hrp = player.Character:FindFirstChild("HumanoidRootPart") or player.Character:FindFirstChild("UpperTorso")
    if not hrp then return end

    local distToEntity = (root.Position - hrp.Position).Magnitude
    local currentlyHiding = isPlayerHiding()

    if currentlyHiding and distToEntity < 46 then
        if not warningGuiActive then
            warningGuiActive = true
            showEntityWarning()
        end

        for _, prompt in ipairs(workspace:GetDescendants()) do
            if prompt:IsA("ProximityPrompt") then
                local actionText = prompt.ActionText and prompt.ActionText:lower() or ""
                local objectText = prompt.ObjectText and prompt.ObjectText:lower() or ""
                if string.find(actionText, "exit") or string.find(actionText, "leave") or string.find(objectText, "wardrobe") or string.find(objectText, "closet") then
                    prompt.Enabled = false
                    task.delay(0.2, function()
                        if prompt then prompt.Enabled = true end
                    end)
                end
            end
        end
    else
        if distToEntity >= 50 then
            warningGuiActive = false
        end
    end
end)

---====== Mechanics (Strictly Ignoring Hiding Rooms) ======---

local isMoving = false

local function destructPath(originCFrame, radius)
    local originPos = originCFrame.Position
    local moveDirection = originCFrame.LookVector

    local region = OverlapParams.new()
    region.FilterType = Enum.RaycastFilterType.Exclude

    local filterList = {}
    if entity.Model then
        table.insert(filterList, entity.Model)
    end
    if Players.LocalPlayer.Character then
        table.insert(filterList, Players.LocalPlayer.Character)
    end
    region.FilterDescendantsInstances = filterList

    local parts = workspace:GetPartBoundsInRadius(originPos, radius, region)
    local soundCooldown = false

    for _, part in ipairs(parts) do
        if part:IsA("BasePart") then
            if not isBlacklisted(part) and not isGuidingLightBook(part) then
                local parentModel = part.Parent
                if parentModel and parentModel:IsA("Model") and string.find(parentModel.Name:lower(), "bookshelf") then
                    processBookshelf(parentModel, moveDirection)
                else
                    if part.Anchored then
                        part.Anchored = false
                        if not soundCooldown then
                            soundCooldown = true
                            playRandomBreakSound(part.Position, 6.0)
                            task.delay(0.1, function() soundCooldown = false end)
                        end
                    end

                    local throwVelocity = (moveDirection + Vector3.new(math.random(-2, 2)/10, 0.35, math.random(-2, 2)/10)).Unit * math.random(130, 200)
                    part.AssemblyLinearVelocity = throwVelocity
                end
            end
        end
    end
end

---====== Callbacks ======---

local hasTriggeredJumpscare = false

local function triggerJumpscareSequence()
    if hasTriggeredJumpscare then return end
    hasTriggeredJumpscare = true

    local player = Players.LocalPlayer
    if not player or not player.Character then return end

    local char = player.Character
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if hrp then hrp.Anchored = true end

    -- Loop through all parts in the character and apply meat texture + random faces to any body parts (Torso/UpperTorso/LowerTorso/Head/etc.)
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") then
            -- Optional: Color parts red to fit the meat theme
            part.Color = Color3.fromRGB(150, 0, 0)

            -- Apply meat texture to all major faces so it completely covers the body
            for _, normalId in ipairs(Enum.NormalId:GetEnumItems()) do
                local meat = Instance.new("Decal")
                meat.Texture = meatTextureId
                meat.Face = normalId
                meat.Parent = part

                local face = Instance.new("Decal")
                face.Texture = jumpscareFaces[math.random(1, #jumpscareFaces)]
                face.Face = normalId
                face.Parent = part
            end
        end
    end

    local chosenFace = jumpscareFaces[math.random(1, #jumpscareFaces)]

    local screamSound = Instance.new("Sound")
    screamSound.SoundId = SCREAM_SOUND_ID
    screamSound.Volume = 10
    screamSound.Parent = SoundService
    screamSound:Play()
    Debris:AddItem(screamSound, 5)

    local playerGui = player:FindFirstChild("PlayerGui")
    if playerGui then
        local screenGui = Instance.new("ScreenGui")
        screenGui.Name = "A60ScreenJumpscare"
        screenGui.IgnoreGuiInset = true
        screenGui.DisplayOrder = 999999
        screenGui.ResetOnSpawn = false
        screenGui.Parent = playerGui

        local redFrame = Instance.new("Frame")
        redFrame.Size = UDim2.new(1, 0, 1, 0)
        redFrame.BackgroundColor3 = Color3.fromRGB(255, 0, 0)
        redFrame.BackgroundTransparency = 0.3
        redFrame.BorderSizePixel = 0
        redFrame.Parent = screenGui

        local faceImage = Instance.new("ImageLabel")
        faceImage.Size = UDim2.new(1, 0, 1, 0)
        faceImage.BackgroundTransparency = 1
        faceImage.ScaleType = Enum.ScaleType.Fit
        faceImage.Parent = screenGui

        task.spawn(function()
            for i = 1, 25 do
                faceImage.Image = jumpscareFaces[math.random(1, #jumpscareFaces)]
                task.wait(0.03)
            end

            faceImage.Image = chosenFace

            local stretchTween = TweenService:Create(faceImage, TweenInfo.new(3.0, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                Size = UDim2.new(1.4, 0, 1.4, 0),
                Position = UDim2.new(-0.2, 0, -0.2, 0)
            })
            stretchTween:Play()

            local fadeTween = TweenService:Create(redFrame, TweenInfo.new(3.5, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                BackgroundTransparency = 1
            })
            local imageFadeTween = TweenService:Create(faceImage, TweenInfo.new(3.5, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                ImageTransparency = 1
            })

            fadeTween:Play()
            imageFadeTween:Play()

            imageFadeTween.Completed:Connect(function()
                screenGui:Destroy()
            end)
        end)
    end
end

entity:SetCallback("OnStartMoving", function()
    isMoving = true
    hasTriggeredJumpscare = false

    local connection
    connection = RunService.RenderStepped:Connect(function()
        if not isMoving or hasTriggeredJumpscare then
            if connection then connection:Disconnect() end
            return
        end

        local model = entity.Model
        if model then
            local root = model.PrimaryPart or model:FindFirstChildWhichIsA("BasePart")
            if root then
                local origin = root.CFrame
                destructPath(origin, 45)

                local player = Players.LocalPlayer
                if player and player.Character then
                    local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
                    if humanoid and humanoid.Health > 0 then
                        local hrp = player.Character:FindFirstChild("HumanoidRootPart") or player.Character:FindFirstChild("UpperTorso")
                        if hrp then
                            if not isPlayerHiding() then
                                local dist = (root.Position - hrp.Position).Magnitude
                                if dist <= 46 then
                                    connection:Disconnect()
                                    triggerJumpscareSequence()
                                end
                            end
                        end
                    end
                end
            end
        end
    end)
end)

entity:SetCallback("OnDespawning", function()
    isMoving = false
end)

entity:SetCallback("OnDespawned", function()
    isMoving = false
end)

---====== Run Entity ======---

entity:Run()
